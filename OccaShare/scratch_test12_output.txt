indow.showSuccess('Booking updated successfully.');
            // Refresh modal
            window.openBookingDetailModal(id);
        }
    } catch(err) {
        alert("Error saving booking: " + err);
    } finally {
        btn.disabled = false;
        btn.innerHTML = 'Save Changes';
    }
