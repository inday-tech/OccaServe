    db: Session = Depends(database.get_db),
    user: models.User = Depends(caterer_only)
):
    profile = user.caterer_profile
    bookings = [b for b in profile.bookings if b.status not in ['draft', 'pending_quotation', 'pending_review', 'inquiry', 'negotiating', 'quoted'] and not b.is_archived]

    today = date.today()

    upcoming_events_count = len([b for b in all_bookings if b.event_date and today < b.event_date <= today + timedelta(days=30) and b.status not in ['cancelled', 'draft']])

    outstanding_count = 0
    action_center_items = []

        if b.status in ['cancelled', 'draft', 'completed']:
            continue
            
        if b.booking_source == 'Internal':
            if b.special_requests == 'pinned':
                action_center_items.append({'type': 'warning', 'title': 'Pinned Schedule', 'desc': f'{b.event_name} on {b.event_date.strftime("%b %d")}', 'icon': 'fa-thumbtack'})
            continue
            
        amount = float(b.total_amount or b.total_price or 0)
        paid = float(b.amount_paid or 0)

            outstanding_balance += (amount - paid)
            outstanding_count += 1
